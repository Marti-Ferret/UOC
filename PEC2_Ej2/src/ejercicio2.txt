interface Plane{
    model:string,
    npassengers:number
}
interface HangarHash {
    [id: string]: Plane
}

let myHangar:HangarHash = {}

myHangar['123Z']={
    model:'airbus',
    npassengers:200
}
myHangar['H789']={ 
    model:'boeing',
    npassengers:151
}

for (let id in myHangar) {
    console.log(id + ":" +
        myHangar[id].model +
        "(" + myHangar[id].npassengers + ")"
    )
}
